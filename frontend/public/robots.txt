# Equip — public learning platform.
# Crawl the public surface; do not crawl the API or authenticated
# routes (they are gated server-side and return JSON, not HTML).

# AI crawlers — search (OAI-SearchBot, PerplexityBot, Claude-SearchBot) and
# training (GPTBot, ClaudeBot, Google-Extended, CCBot) alike — are not named
# here on purpose: the `*` group below covers them. Equip's courses are free
# and meant to be found, and an answer engine that has read them can point a
# reader here. A summary for them is at /llms.txt.

User-agent: *
Allow: /
# Public documents that happen to sit under a disallowed prefix. Robots.txt
# matches on prefix, so `Disallow: /teacher` would otherwise hide the Teacher
# & Contributor Agreement — a document somebody is asked to accept and is
# entitled to find before they are standing in front of the checkbox.
Allow: /teacher-terms
Disallow: /admin
Disallow: /teacher
Disallow: /dashboard
Disallow: /profile
Disallow: /certificates
Disallow: /calendar
Disallow: /notes
# One-time links carrying a token in the URL. Nothing to index, and a
# crawler following one burns it.
Disallow: /auth
# A personal link from the foot of a course mail; it asks before it acts.
Disallow: /unsubscribe
# Sign-in, sign-up and password-reset forms: nothing on them to find.
Disallow: /login
Disallow: /register
Disallow: /forgot-password
Disallow: /invite

Sitemap: https://equipbible.com/sitemap.xml
